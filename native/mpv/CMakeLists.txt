cmake_minimum_required(VERSION 3.21)
project(liquid_media LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick OpenGL DBus)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MPV REQUIRED IMPORTED_TARGET mpv)
pkg_check_modules(EGL REQUIRED IMPORTED_TARGET egl)
add_library(liquidmedia MODULE liquidmedia.cpp mpris.cpp mpris.h)
target_link_libraries(liquidmedia PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::OpenGL Qt6::DBus PkgConfig::MPV PkgConfig::EGL)
target_compile_options(liquidmedia PRIVATE -Wall -Wextra -Werror)
option(LIQUID_MEDIA_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(LIQUID_MEDIA_SANITIZERS)
  target_compile_options(liquidmedia PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(liquidmedia PRIVATE -fsanitize=address,undefined)
endif()
install(TARGETS liquidmedia LIBRARY DESTINATION minerva-shell/native/Liquid/Media)
install(FILES qmldir DESTINATION minerva-shell/native/Liquid/Media)
