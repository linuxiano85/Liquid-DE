cmake_minimum_required(VERSION 3.21)
project(minerva_window_identity LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Gui GuiPrivate Qml Quick WaylandClientPrivate)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client)
add_library(minervawindowidentity MODULE windowidentity.cpp)
target_link_libraries(minervawindowidentity PRIVATE Qt6::Core Qt6::Gui Qt6::GuiPrivate Qt6::Qml Qt6::Quick Qt6::WaylandClientPrivate PkgConfig::WAYLAND)
target_compile_options(minervawindowidentity PRIVATE -Wall -Wextra -Werror)
install(TARGETS minervawindowidentity LIBRARY DESTINATION minerva-shell/native/Minerva/WindowIdentity)
