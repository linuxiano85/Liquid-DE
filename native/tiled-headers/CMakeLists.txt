cmake_minimum_required(VERSION 3.20)
project(minerva_tiled_headers LANGUAGES C)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED gtk4)
pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET gio-2.0)
add_library(minervatiledheaders SHARED tiledheaders.c)
target_include_directories(minervatiledheaders PRIVATE ${GTK_INCLUDE_DIRS})
target_link_libraries(minervatiledheaders PRIVATE PkgConfig::GLIB ${CMAKE_DL_LIBS})
target_compile_options(minervatiledheaders PRIVATE -Wall -Wextra -Werror)
install(TARGETS minervatiledheaders LIBRARY DESTINATION lib/minerva)
